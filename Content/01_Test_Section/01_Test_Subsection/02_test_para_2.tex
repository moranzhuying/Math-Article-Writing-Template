\begin{exercise}{测试习题一}{exr:test-1}
    设 $A$ 为 $n$ 阶方阵,证明 $\tr(A^{\mathsf T}) = \tr(A)$.
\end{exercise}

\begin{solution}
    转置保持对角元素不变,故迹相等.
\end{solution}

\begin{exercise}{测试习题二}{exr:test-2}
    求极限 $\displaystyle\lim_{x\to 0}\frac{\sin x}{x}$.
\end{exercise}

\begin{solution}
    由重要极限知其值为 $1$.
\end{solution}
